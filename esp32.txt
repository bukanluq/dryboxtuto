#include <DHT11.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <WiFi.h>
#include <WebServer.h>
#include <Preferences.h>
#include <HTTPClient.h>
#include <WiFiClientSecure.h>

LiquidCrystal_I2C led(0x27,16,2);
// DHT11 connected to ESP32 GPIO 4
DHT11 dht11(4);

// Humidifier relay/transistor connected to ESP32 GPIO 5
const int humidifierPin = 5;

const int fanPin = 2;

bool isDoorOpen = false;

// Mirrors exactly what the control logic below decided, so the cloud
// status always agrees with what Serial Monitor prints.
bool humidifierIsOn = false;

/* ---------------------------------------------------------
   SUPABASE CONFIG - fill these in (same project as the webapp)
--------------------------------------------------------- */
const char* SUPABASE_URL = "YOUR_SUPABASE_PROJECT_URL";    // e.g. "https://xxxxxxxx.supabase.co"
const char* SUPABASE_ANON_KEY = "YOUR_SUPABASE_ANON_KEY";  // Project Settings -> API -> anon public key
const char* STATUS_TABLE = "device_status";

unsigned long lastPushMs = 0;
const unsigned long PUSH_INTERVAL_MS = 5000; // push status every 5s - plenty for humidity readings

/* ---------------------------------------------------------
   DEVICE IDENTITY
   Every board gets a fixed, unique ID derived from its own
   chip (no two ESP32s share one), so you can run several
   devices and tell them apart in the webapp.
--------------------------------------------------------- */

String DEVICE_ID = ""; // e.g. "A1B2C3" - computed once in setup()

Preferences preferences;

void computeDeviceId() {
    uint64_t mac = ESP.getEfuseMac();
    char idBuf[7];
    sprintf(idBuf, "%06X", (uint32_t)(mac & 0xFFFFFFUL));
    DEVICE_ID = String(idBuf);
}

/* ---------------------------------------------------------
   PUSH STATUS TO SUPABASE
   Upserts one row per device (keyed by device_id) so the
   table stays at a fixed size no matter how often this runs.
   The webapp reads this row directly - no connection to the
   ESP32 itself is needed, so it works from any phone/network.
--------------------------------------------------------- */

void pushStatusToSupabase() {
    if (WiFi.status() != WL_CONNECTED) return;

    WiFiClientSecure client;
    client.setInsecure(); // skip certificate validation - simplest option for a hobby project

    HTTPClient http;
    String url = String(SUPABASE_URL) + "/rest/v1/" + STATUS_TABLE + "?on_conflict=device_id";

    if (!http.begin(client, url)) {
        Serial.println("Supabase push: failed to begin request");
        return;
    }

    http.addHeader("apikey", SUPABASE_ANON_KEY);
    http.addHeader("Authorization", String("Bearer ") + SUPABASE_ANON_KEY);
    http.addHeader("Content-Type", "application/json");
    http.addHeader("Prefer", "resolution=merge-duplicates,return=minimal");

    String json = "{";
    json += "\"device_id\":\"" + DEVICE_ID + "\",";
    json += "\"humidity\":" + String(lastHumidityValid ? lastHumidity : -1) + ",";
    json += "\"humidity_valid\":" + String(lastHumidityValid ? "true" : "false") + ",";
    json += "\"humidifier_on\":" + String(humidifierIsOn ? "true" : "false") + ",";
    json += "\"fan_on\":" + String(isDoorOpen ? "true" : "false");
    json += "}";

    int httpCode = http.POST(json);

    if (httpCode > 0 && httpCode < 300) {
        Serial.println("Supabase push OK");
    } else {
        Serial.print("Supabase push failed, code: ");
        Serial.println(httpCode);
        Serial.println(http.getString());
    }

    http.end();
}

/* ---------------------------------------------------------
   WIFI PROVISIONING
   - On boot, tries to connect using credentials saved in
     flash (Preferences / NVS).
   - If none are saved, or the connection fails, the ESP32
     starts its own Access Point with a setup web page.
   - Saving reboots the board, which then connects using the
     new saved credentials.
--------------------------------------------------------- */

WebServer server(80);

const char* AP_SSID_PREFIX = "ESP32-Setup-"; // full AP name includes the device ID, e.g. ESP32-Setup-A1B2C3
const char* AP_PASSWORD = "";                 // "" = open AP. Set a password (min 8 chars) to secure it.

bool apMode = false;
const unsigned long WIFI_CONNECT_TIMEOUT_MS = 15000;

unsigned long lastSensorReadMs = 0;
const unsigned long SENSOR_READ_INTERVAL_MS = 1000;

int lastHumidity = -1;
bool lastHumidityValid = false;

/* ---------------------------------------------------------
   LOCAL STATUS PAGE (handy for debugging on the same network -
   the webapp itself no longer needs this)
--------------------------------------------------------- */

void handleStatusPage() {
    String html = "<!DOCTYPE html><html><head><meta name='viewport' content='width=device-width, initial-scale=1'>";
    html += "<title>ESP32 Monitor</title></head><body style='font-family:Arial;padding:20px;'>";
    html += "<h2>Device Online</h2>";
    html += "<p>Device ID: <b>" + DEVICE_ID + "</b></p>";
    html += "<p>Humidity: " + String(lastHumidityValid ? String(lastHumidity) + " %" : "--") + "</p>";
    html += "<p>Status pushes to the cloud every " + String(PUSH_INTERVAL_MS / 1000) + "s automatically.</p>";
    html += "<p><a href='/forget-wifi' style='color:#d9483b;'>Forget WiFi &amp; reset</a></p>";
    html += "</body></html>";
    server.send(200, "text/html", html);
}

void handleForgetWifiPage() {
    String html = "<!DOCTYPE html><html><head><meta name='viewport' content='width=device-width, initial-scale=1'>";
    html += "<title>Reset Device</title></head><body style='font-family:Arial;padding:20px;'>";
    html += "<h2>Forget WiFi & Reset</h2>";
    html += "<p>This erases the saved WiFi network, then restarts into setup mode (AP: <b>" + String(AP_SSID_PREFIX) + DEVICE_ID + "</b>).</p>";
    html += "<form action='/forget-wifi' method='POST'>";
    html += "<input type='submit' value='Forget WiFi & Restart' style='padding:10px 20px;background:#d9483b;color:#fff;border:none;'>";
    html += "</form>";
    html += "</body></html>";
    server.send(200, "text/html", html);
}

void handleForgetWifi() {
    preferences.begin("wifi", false);
    preferences.clear();
    preferences.end();

    server.send(200, "text/html",
        "<html><body style='font-family:Arial;padding:20px;'>"
        "<h3>WiFi cleared. Restarting into setup mode...</h3></body></html>");

    delay(1500);
    ESP.restart();
}

void handleRoot() {
    if (!apMode) {
        handleStatusPage();
        return;
    }

    int networkCount = WiFi.scanNetworks();

    String html = "<!DOCTYPE html><html><head><meta name='viewport' content='width=device-width, initial-scale=1'>";
    html += "<title>Device Setup</title></head><body style='font-family:Arial;padding:20px;'>";
    html += "<h2>Set Up This Device</h2>";
    html += "<p>Device ID: <b>" + DEVICE_ID + "</b> - enter this ID in the monitoring webapp when you register this item.</p>";
    html += "<form action='/save' method='POST'>";

    html += "<label>WiFi Network:</label><br><select name='ssid' style='width:100%;padding:8px;margin:8px 0;'>";
    if (networkCount == 0) {
        html += "<option value=''>No networks found - rescan</option>";
    } else {
        for (int i = 0; i < networkCount; i++) {
            html += "<option value='" + WiFi.SSID(i) + "'>";
            html += WiFi.SSID(i) + " (" + String(WiFi.RSSI(i)) + " dBm)";
            html += "</option>";
        }
    }
    html += "</select><br>";

    html += "<label>WiFi Password:</label><br>";
    html += "<input type='password' name='wifiPassword' style='width:100%;padding:8px;margin:8px 0;'><br>";

    html += "<hr style='margin:20px 0;'>";

    html += "<label>Keep a Fixed IP (optional):</label><br>";
    html += "<div style='color:#888;font-size:13px;margin-bottom:6px;'>Not required anymore - the webapp reads from the cloud now. Only useful if you still want to visit this device's local status page at a consistent address.</div>";
    html += "<input type='number' name='staticOctet' min='2' max='254' placeholder='e.g. 50' style='width:100%;padding:8px;margin:8px 0;'><br>";

    html += "<input type='submit' value='Save & Connect' style='padding:10px 20px;margin-top:10px;'>";
    html += "</form>";
    html += "<p style='color:#888;font-size:13px;'>Setup AP: " + String(AP_SSID_PREFIX) + DEVICE_ID + " | IP: " + WiFi.softAPIP().toString() + "</p>";
    html += "</body></html>";

    server.send(200, "text/html", html);
}

void handleSave() {
    String ssid = server.arg("ssid");
    String wifiPassword = server.arg("wifiPassword");
    String staticOctet = server.arg("staticOctet");

    if (ssid.length() == 0) {
        server.send(400, "text/plain", "No network selected. Go back and try again.");
        return;
    }

    preferences.begin("wifi", false);
    preferences.putString("ssid", ssid);
    preferences.putString("password", wifiPassword);
    preferences.putString("octet", staticOctet); // empty string means "no fixed IP requested"
    preferences.end();

    server.send(200, "text/html",
        "<html><body style='font-family:Arial;padding:20px;'>"
        "<h3>Saved. Rebooting and connecting...</h3>"
        "<p>Remember this device's ID - you'll need it to register this item in the monitoring webapp.</p>"
        "</body></html>");

    delay(2000);
    ESP.restart();
}

void startServer() {
    server.on("/", handleRoot);
    server.on("/save", HTTP_POST, handleSave);
    server.on("/forget-wifi", HTTP_GET, handleForgetWifiPage);
    server.on("/forget-wifi", HTTP_POST, handleForgetWifi);
    server.begin();
}

void startAPMode() {
    apMode = true;

    WiFi.mode(WIFI_AP);
    String apName = String(AP_SSID_PREFIX) + DEVICE_ID;
    WiFi.softAP(apName.c_str(), strlen(AP_PASSWORD) ? AP_PASSWORD : NULL);

    startServer();

    Serial.println("AP mode started.");
    Serial.print("Connect to WiFi: ");
    Serial.println(apName);
    Serial.print("Then open: http://");
    Serial.println(WiFi.softAPIP());

    led.clear();
    led.setCursor(0,0);
    led.print("ID: " + DEVICE_ID);
    led.setCursor(0,1);
    led.print(WiFi.softAPIP());
}

bool connectOnce(const String& ssid, const String& password, unsigned long timeoutMs) {
    WiFi.begin(ssid.c_str(), password.c_str());

    unsigned long start = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - start < timeoutMs) {
        delay(250);
        Serial.print(".");
    }
    Serial.println();

    return WiFi.status() == WL_CONNECTED;
}

bool connectWithSavedCredentials() {
    preferences.begin("wifi", true);
    String ssid = preferences.getString("ssid", "");
    String password = preferences.getString("password", "");
    String staticOctetStr = preferences.getString("octet", "");
    preferences.end();

    if (ssid.length() == 0) {
        Serial.println("No saved WiFi credentials found.");
        return false;
    }

    Serial.print("Connecting to saved network: ");
    Serial.println(ssid);

    WiFi.mode(WIFI_STA);

    if (!connectOnce(ssid, password, WIFI_CONNECT_TIMEOUT_MS)) {
        Serial.println("Failed to connect to saved network.");
        return false;
    }

    if (staticOctetStr.length() > 0) {
        int octet = staticOctetStr.toInt();

        if (octet >= 2 && octet <= 254) {
            IPAddress gateway = WiFi.gatewayIP();
            IPAddress subnet = WiFi.subnetMask();
            IPAddress dns = WiFi.dnsIP();
            IPAddress staticIP(gateway[0], gateway[1], gateway[2], octet);

            Serial.print("Requesting fixed IP: ");
            Serial.println(staticIP);

            WiFi.disconnect(true, false);
            delay(300);

            WiFi.config(staticIP, gateway, subnet, dns);

            if (!connectOnce(ssid, password, 10000)) {
                Serial.println("Fixed IP failed, falling back to automatic address.");
                WiFi.config(IPADDR_NONE, IPADDR_NONE, IPADDR_NONE);
                connectOnce(ssid, password, WIFI_CONNECT_TIMEOUT_MS);
            }
        }
    }

    if (WiFi.status() == WL_CONNECTED) {
        Serial.print("Connected. IP: ");
        Serial.println(WiFi.localIP());

        led.clear();
        led.setCursor(0,0);
        led.print("ID: " + DEVICE_ID);
        led.setCursor(0,1);
        led.print(WiFi.localIP());
        delay(2000);
        led.clear();

        return true;
    }

    return false;
}

void setupWiFi() {
    if (connectWithSavedCredentials()) {
        apMode = false;
        startServer();
    } else {
        startAPMode();
    }
}

/* ---------------------------------------------------------
   SENSOR / HUMIDIFIER CONTROL (unchanged logic)
--------------------------------------------------------- */

void readAndControlHumidity() {
    int humidity = dht11.readHumidity();

    led.setCursor(0,1);
    led.print("RH: ");
    led.print(humidity);
    led.print("%   ");
    led.setCursor(0,0);
    led.print("ID: " + DEVICE_ID);

    if (humidity != DHT11::ERROR_CHECKSUM && humidity != DHT11::ERROR_TIMEOUT) {
        Serial.print("Humidity: ");
        Serial.print(humidity);
        Serial.println(" %");
        lastHumidity = humidity;
        lastHumidityValid = true;
    } else {
        Serial.println(DHT11::getErrorString(humidity));
        lastHumidityValid = false;
        return;
    }

    if (humidity < 45) {
        if (isDoorOpen) {
            digitalWrite(fanPin, LOW);
            isDoorOpen = false;
        }
        Serial.print("Door is closed");
        digitalWrite(humidifierPin, HIGH);
        humidifierIsOn = true;
        Serial.println(" | Humidifier is ON");
    }
    else if (humidity > 55) {
        if (!isDoorOpen) {
            digitalWrite(fanPin, HIGH);
            isDoorOpen = true;
        }
        Serial.print("Door is opened");
        digitalWrite(humidifierPin, LOW);
        humidifierIsOn = false;
        Serial.println(" | Humidifier is OFF");
    }
    else {
        if (isDoorOpen) {
            digitalWrite(fanPin, LOW);
            isDoorOpen = false;
        }
        Serial.print("Door is closed");
        digitalWrite(humidifierPin, LOW);
        humidifierIsOn = false;
        Serial.println(" | Humidifier is OFF");
    }
}

void setup() {
    Serial.begin(115200);
    pinMode(humidifierPin, OUTPUT);
    pinMode(fanPin, OUTPUT);
    led.init();
    led.backlight();

    computeDeviceId();

    Serial.print("Device ID: ");
    Serial.println(DEVICE_ID);

    setupWiFi();
}

void loop() {
    server.handleClient();

    unsigned long now = millis();

    if (now - lastSensorReadMs >= SENSOR_READ_INTERVAL_MS) {
        lastSensorReadMs = now;
        readAndControlHumidity();
    }

    if (!apMode && now - lastPushMs >= PUSH_INTERVAL_MS) {
        lastPushMs = now;
        pushStatusToSupabase();
    }
}
